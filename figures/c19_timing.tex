% The coherency figure: three moments of one transfer, cache row above,
% memory row below.
\begin{tikzpicture}[font=\sffamily\scriptsize]

\foreach \x/\t in {0/{1. the processor writes the sentinel},
                   46/{2. the engine writes the burst},
                   92/{3. the processor reads}}{
  \node[font=\sffamily\bfseries\scriptsize, anchor=north west, text width=40mm]
    at (\x mm,92mm) {\t};}
\foreach \x in {38,84}{\draw[tick] (\x mm,30mm) -- (\x mm,68mm);}

\node[signame, anchor=east] at (-2mm,58mm) {D-cache};
\node[signame, anchor=east] at (-2mm,40mm) {SRAM};

% ---- moment 1
\node[master, text width=22mm, minimum height=7mm] (cpu1) at (12mm,70mm) {processor};
\draw[cpuflow] (12mm,66mm) -- (12mm,61mm);
\foreach \x in {0,8,16}{\cacheline{\x}{56}{cwdirty}{A5}}
\foreach \x in {0,8,16}{\cacheline{\x}{38}{cacheword}{old}}

% ---- moment 2
\foreach \x in {46,54,62}{\cacheline{\x}{56}{cwstale}{A5}}
\foreach \x/\v in {46/1F, 54/22, 62/03}{\cacheline{\x}{38}{cacheword}{\v}}
\node[master, text width=26mm, minimum height=7mm] (eng) at (58mm,24mm)
  {transfer engine};
\draw[dmaflow] (58mm,28mm) -- (58mm,37mm);
\draw[incoherent] (46mm,50mm) -- (70mm,50mm);
\node[font=\tiny, text=red!70!black, anchor=west] at (71mm,50mm) {no link};

% ---- moment 3
\foreach \x in {92,100,108}{\cacheline{\x}{56}{cwstale}{A5}}
\foreach \x/\v in {92/1F, 100/22, 108/03}{\cacheline{\x}{38}{cacheword}{\v}}
\draw[incoherent] (92mm,50mm) -- (116mm,50mm);
\draw[cpuflow] (100mm,61mm) -- (100mm,68mm);
\node[font=\tiny, anchor=west, text width=32mm] at (102mm,66mm)
  {naive: the read hits the stale line and returns A5};
\draw[dmaflow] (117mm,40mm) -- (124mm,40mm) -- (124mm,48mm);
\node[font=\tiny, anchor=west, text width=26mm] at (102mm,32mm)
  {after invalidate: the read reaches SRAM};

% ---- legend
\cacheline{0}{8}{cwclean, minimum width=14mm}{clean}
\cacheline{18}{8}{cwdirty, minimum width=14mm}{dirty}
\cacheline{36}{8}{cwstale, minimum width=14mm}{stale}
\node[font=\tiny, anchor=west, text width=76mm] at (54mm,10mm)
  {clean: agrees with memory. dirty: the processor wrote and memory has not
   caught up yet. stale: memory moved on and the line did not.};

\node[umlnote, text width=44mm, anchor=north west] at (0mm,0mm)
  {The zigzag is the point: there is no link between the two rows on this core.
   The engine neither fills the cache nor looks in it, so nothing in the
   hardware makes the rows agree.};
\node[umlnote, text width=44mm, anchor=north west] at (48mm,0mm)
  {Invalidating by address before the read drops the stale line, so the read
   goes to SRAM. That is the first of the three fixes and the cheapest.};
\node[umlnote, text width=44mm, anchor=north west] at (96mm,0mm)
  {Invalidating a buffer that shares its first or last line with a neighbour
   drops the neighbour's pending writes too. Align to 32 bytes and size to
   whole lines.};
\end{tikzpicture}
