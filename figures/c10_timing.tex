\begin{tikzpicture}[font=\sffamily\scriptsize]

  \draw[tick] (\x mm,2mm) -- (\x mm,64mm);}

% ---- the current trace (shape only: no number is claimed)
\draw[signal]
  (0mm,42mm) -- (15mm,42mm) -- (15mm,54mm) -- (28mm,54mm) -- (28mm,48mm)
  -- (48mm,48mm) -- (48mm,52mm) -- (60mm,52mm) -- (60mm,58mm) -- (82mm,58mm)
  -- (82mm,42mm) -- (100mm,42mm);
\node[signame] at (0mm,50mm) {current};
\node[font=\sffamily\tiny, anchor=west] at (101mm,42mm) {sleep floor};

% ---- the three marker lanes, D2 is the high bit
\draw[signalb]
  (0mm,30mm) -- (60mm,30mm) -- (60mm,34mm) -- (82mm,34mm) -- (82mm,30mm) -- (100mm,30mm);
\node[signame] at (0mm,32mm) {marker D2};

  (0mm,20mm) -- (28mm,20mm) -- (28mm,24mm) -- (60mm,24mm) -- (60mm,20mm) -- (100mm,20mm);

  (0mm,10mm) -- (15mm,10mm) -- (15mm,14mm) -- (28mm,14mm) -- (28mm,10mm)
  -- (48mm,10mm) -- (48mm,14mm) -- (60mm,14mm) -- (60mm,10mm) -- (100mm,10mm);

% ---- phase names and codes

\node[umlnote, text width=48mm, anchor=north west] at (54mm,-2mm)

\end{tikzpicture}
