% The node's state machine, and the hysteresis inside its steady state.
\begin{tikzpicture}[font=\sffamily\scriptsize]

% ---------------- operating states
\node[initial] (i) at (0,0) {};
\node[state] (cold) at (1.9,0) {cold};
\node[state] (up) at (5.4,0) {uploading};
\node[state] (rng) at (9.4,0) {ranging};
\node[state] (re) at (13.4,0) {re-upload};

\draw[trans] (i) -- (cold);
\draw[trans] (cold) -- node[above]{identity ok} (up);
\draw[trans] (up) -- node[above]{upload done} (rng);
\draw[trans] (rng) -- node[above]{sensor silent} (re);
\draw[trans] (re) to[out=110,in=70] node[above]{reset and retry} (up);

% ---------------- the hysteresis inside ranging
\begin{scope}[yshift=-38mm]
\node[rtstate] (cl) at (2.0,0) {clear};
\node[rtstate] (ap) at (6.2,0) {approaching};
\node[rtstate] (pr) at (10.4,0) {present};
\node[rtstate] (lv) at (14.0,0) {leaving};
\draw[trans] (cl) -- node[above, align=center]{under\\1500 mm} (ap);
\draw[trans] (ap) -- node[above, align=center]{under 800 mm\\3 of 5 frames} (pr);
\draw[trans] (pr) -- node[above, align=center]{over 950 mm\\3 of 5} (lv);
\draw[trans] (lv) to[out=-110,in=-70] node[below]{over 1500 mm} (cl);
\begin{scope}[on background layer]
\node[layer, fit=(cl)(ap)(pr)(lv), inner ysep=17pt, inner xsep=6pt,
      label={[layerlabel, anchor=south west]north west:inside ranging}] (hyst) {};
\end{scope}
\end{scope}

\draw[dotted, draw=linecol, thick] (rng.south) -- (hyst.north -| rng.south);

\node[umlnote, text width=52mm, anchor=north west] at (0,-8.2)
  {The re-upload state is the one most implementations omit. A supply
   disturbance can empty the sensor without resetting the board; without this
   state the node reports the last distance it saw, for ever.};
\node[umlnote, text width=52mm, anchor=north west] at (8.6,-8.2)
  {Two thresholds and a confirmation count, never one threshold. The numbers are
   a starting point measured against a real target rather than constants to be
   trusted. The status byte carries the state and its confidence.};
\end{tikzpicture}
