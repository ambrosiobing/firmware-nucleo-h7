% Two exchanges on the serial link: an unsolicited message mid-command, and a
% reply that arrives after its deadline.
\begin{tikzpicture}[font=\sffamily\scriptsize]

% ---- idle lines
\draw[signal] (0mm,40mm) -- (136mm,40mm);
\draw[signal] (0mm,24mm) -- (136mm,24mm);
\node[signame] at (-1mm,40mm) {to module};
\node[signame] at (-1mm,24mm) {from module};
\draw[taxis] (0mm,8mm) -- (140mm,8mm) node[right, font=\tiny] {time};

% ---- exchange one
\busval{40}{0}{18}{AT+CSQ}
\busval{24}{22}{20}{+CEREG: 1}
\busval{24}{46}{22}{+CSQ: nn,nn}
\busval{24}{72}{10}{OK}

% ---- exchange two
\busval{40}{90}{22}{AT+CPIN?}
\busval{24}{120}{14}{OK}
\draw[deadline] (112mm,14mm) -- (112mm,46mm);
\node[font=\sffamily\tiny, anchor=south, align=center] at (112mm,46mm)
  {deadline from the manual};

% ---- measured spans
\draw[tspan] (0mm,16mm) -- (82mm,16mm);
\node[tlabel] at (41mm,16mm) {one command, start to final result};
\draw[tspan] (90mm,16mm) -- (112mm,16mm);
\node[tlabel] at (101mm,16mm) {timeout window};

% ---- annotations, staggered so no two labels share a band
\node[font=\sffamily\tiny, anchor=south, align=center] at (32mm,26mm)
  {unsolicited: dispatched from the table};
\node[font=\sffamily\tiny, anchor=south, align=center] at (57mm,32mm)
  {intermediate: belongs to the command};
\node[font=\sffamily\tiny, anchor=south, align=center] at (77mm,26mm)
  {final: completes it};
\node[font=\sffamily\tiny, anchor=south, align=center] at (127mm,30mm)
  {late: discarded};

\node[umlnote, text width=52mm, anchor=north west] at (0mm,4mm)
  {A blocking driver reads the unsolicited message as the reply to the command
   in flight. The classifier decides the class before the state machine sees
   the line, which removes that failure rather than working around it.};
\node[umlnote, text width=52mm, anchor=north west] at (62mm,4mm)
  {The late reply is discarded and not attached to the following command. Once
   an engine is one command out of step it stays that way, and every later
   result is wrong in a way that looks like a module fault.};
\end{tikzpicture}
