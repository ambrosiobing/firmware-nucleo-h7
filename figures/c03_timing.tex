% Exception numbers on the left; the peripheral IRQ number is in the slot,

% arithmetic everybody gets wrong once. Confirm both in the device header.

\vecentry{7}{vecfree}{}{\ldots}
\vecentry{6}{vecsys}{11}{SVC\_Handler}
\vecentry{5}{vecsys}{14}{PendSV\_Handler}

\vecentry{1}{vecirq}{55}{USART3\_IRQHandler, IRQ 39}

  {Address increases downward. The exception number is on the left; the
