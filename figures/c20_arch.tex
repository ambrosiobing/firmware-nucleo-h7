\draw[irq] (wm.south) -- ++(0,-6mm) -| (sens.north);

  {USART3 result stream\\{\tiny printed from a task, never an ISR}};

  {host: histogram, percentiles, tail plot};
\draw[flow] (link) -- (out);
\draw[flow] (out) -- (host);

\node[umlnote, text width=62mm, anchor=north west] at (0mm,4mm)

\node[umlnote, text width=62mm, anchor=north west] at (68mm,4mm)
