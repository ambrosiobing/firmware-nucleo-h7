% out of the processor.

% ---------------- masters

\node[master, text width=24mm, minimum height=10mm] (bdma) at (128mm,74mm)

% ---------------- caches
\node[cache, text width=18mm, minimum height=8mm] (ic) at (28mm,58mm) {L1 I-cache};

% ---------------- memories on the matrix
\node[memblk, text width=24mm, minimum height=10mm] (fl) at (42mm,28mm) {flash};
\node[memblk, text width=24mm, minimum height=10mm] (axi) at (70mm,28mm) {AXI SRAM};
\node[memblk, text width=24mm, minimum height=10mm] (ahb) at (98mm,28mm) {AHB SRAM};

\node[ext, text width=22mm, minimum height=9mm] (adxl) at (144mm,14mm)
  {ADXL345 breakout};

% ---------------- edges
\draw[cpuflow] (cpu.south -| ic) -- (ic.north);
\draw[cpuflow] (cpu.south -| dc) -- (dc.north);

\draw[cpuflow] (dc.south) -- (dc.south |- mx.north);
\draw[cpuflow] (17mm,74mm) -- (4mm,74mm)

\draw[dmaflow] (dma.south) -- (dma.south |- mx.north);
\draw[dmaflow] (mdma.south) -- (mdma.south |- mx.north);
\draw[dmaflow] (bdma.south) -- (bdma.south |- mx.north);

\draw[bus] (mx.east) -- (spi.west);
\draw[bus] (spi.south) -- (adxl.north);

\node[umlnote, text width=66mm, anchor=north west] at (0mm,4mm)
  {The transfer engines neither fill the cache nor look in it. Two masters,
   two views of the same memory, and no hardware arbitration between them on
   this core. That is the entire mechanism of this chapter.};
\node[umlnote, text width=66mm, anchor=north west] at (74mm,4mm)
  {The main engines cannot reach the tightly coupled memories at all. The
   master engine reaches everything. The low-power engine reaches only the
   small memory in its own domain and keeps running while the rest sleeps.};
\end{tikzpicture}
