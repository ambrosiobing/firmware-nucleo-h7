% The AT engine as a state machine. No arrow leads nowhere.
\begin{tikzpicture}[font=\sffamily\scriptsize, node distance=12mm and 26mm]

\node[initial] (i) at (0,0) {};
\node[state, text width=20mm, right=8mm of i] (idle) {IDLE\\{\tiny queue checked}};
\node[state, text width=20mm, right=of idle] (send) {SENDING\\{\tiny transfer started}};
\node[state, text width=20mm, right=of send] (wait) {WAITING\\{\tiny deadline held}};
\node[state, text width=22mm, below=17mm of wait] (comp) {COMPLETING\\{\tiny callback fires once}};
\node[state, text width=22mm, left=of comp] (back) {BACKOFF\\{\tiny doubled capped jittered}};
\node[state, fill=hwcol, text width=22mm, left=of back] (stop) {STOPPED\\{\tiny NO\_SIM or NO\_MODEM}};
\node[final, left=12mm of stop] (f) {};

\draw[trans] (i) -- (idle);
\draw[trans] (idle) -- node[above]{dequeued} (send);
\draw[trans] (send) -- node[above]{sent} (wait);
\draw[trans] (wait) -- node[right, align=left]{final result\\or deadline} (comp);
\draw[trans] (comp) -- node[above]{retry left} (back);
\draw[trans] (back) -- node[above]{no route} (stop);
\draw[trans] (stop) -- node[above]{report} (f);
% completion returns to idle, routed clear above the top row
\draw[trans] (comp.east) -- ++(10mm,0) |- (13.2,1.6)
  -- node[lbl, pos=0.55, above]{done} (idle.north -| idle) -- (idle.north);
% backoff returns to idle after its interval, routed clear below the bottom row
\draw[trans] (back.south) -- ++(0,-9mm) -|
  node[lbl, pos=0.25, below]{interval elapsed} (idle.south);

\node[umlnote, text width=42mm, anchor=north west] at (-0.4,-4.6)
  {Unsolicited messages are dispatched by the classifier before the state
   machine sees them, in any state, including while a command is in flight.};
\node[umlnote, text width=42mm, anchor=north west] at (4.6,-4.6)
  {Every arrow out of a waiting state is a reply, a deadline or an unsolicited
   message. No amount of backoff produces a SIM card, so the engine reports once
   and stops.};
\node[umlnote, text width=42mm, anchor=north west] at (9.6,-4.6)
  {The completion callback fires exactly once for every submitted command, in
   every outcome, including timeout and abandonment.};
\end{tikzpicture}
