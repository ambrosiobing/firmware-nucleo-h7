% The two free running indices as an event timeline. Not a waveform: the
% vertical axis is a count of bytes, and the gap between the traces is the
% occupancy. Signal timing of real electrical edges is chapters 4 and 6.
\begin{tikzpicture}[font=\sffamily\small]
% ---- grid and axis
\foreach \x in {0,15,...,120}{\draw[tick] (\x mm,-4mm) -- (\x mm,40mm);}
\draw[taxis] (0mm,-4mm) -- (137mm,-4mm);
\node[font=\sffamily\scriptsize, anchor=west] at (138mm,-4mm) {time};
\foreach \x/\t in {0/t0, 15/t1, 30/t2, 45/t3, 60/t4, 75/t5, 90/t6, 105/t7, 120/t8}
  {\node[font=\tiny, anchor=north] at (\x mm,-5mm) {\t};}

% ---- head: written by the producer only
\draw[signalb] (0mm,0mm) -- (15mm,0mm) -- (15mm,6mm) -- (30mm,6mm) -- (30mm,12mm)
  -- (45mm,12mm) -- (45mm,18mm) -- (60mm,18mm) -- (60mm,24mm) -- (90mm,24mm)
  -- (90mm,28mm) -- (105mm,28mm) -- (105mm,32mm) -- (120mm,32mm) -- (120mm,36mm)
  -- (130mm,36mm);
\node[font=\sffamily\scriptsize, anchor=west] at (131mm,36mm) {head};

% ---- tail: written by the consumer only
\draw[signal] (0mm,0mm) -- (30mm,0mm) -- (30mm,2mm) -- (45mm,2mm) -- (45mm,4mm)
  -- (60mm,4mm) -- (60mm,8mm) -- (75mm,8mm) -- (75mm,12mm) -- (90mm,12mm)
  -- (90mm,16mm) -- (105mm,16mm) -- (105mm,20mm) -- (120mm,20mm) -- (120mm,24mm)
  -- (130mm,24mm);
\node[font=\sffamily\scriptsize, anchor=west] at (131mm,24mm) {tail};

% ---- the full window, where bytes are refused
\draw[tspan] (67mm,8mm) -- (67mm,24mm);
\node[tlabel, anchor=west] at (69mm,16mm) {used = 16, full};
\node[sample] at (67mm,24mm) {};
\node[font=\sffamily\tiny, anchor=south] at (67mm,26mm) {bytes refused, drops++};

\draw[tspan] (37mm,0mm) -- (37mm,12mm);
\node[tlabel, anchor=west] at (39mm,6mm) {used = 12};

% ---- the reading of the figure
\node[umlnote, text width=62mm, anchor=north west] at (0mm,-12mm)
  {Both traces only ever rise. The producer owns the upper one and the consumer
   owns the lower one, and neither writes the other. The vertical gap is the
   occupancy, so the buffer is full where the gap reaches the capacity and empty
   where the traces touch.};
\node[umlnote, text width=62mm, anchor=north west] at (70mm,-12mm)
  {The counters wrap only after four thousand million bytes, and the
   subtraction stays correct when they do, because unsigned overflow is defined.
   Nothing in the code treats the wrap as an event.};
\end{tikzpicture}
