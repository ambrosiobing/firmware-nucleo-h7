% The checksum peripheral's control register and the two value registers.
% Only bits 7 down to 0 of the control register carry anything; the rest read
% as zero. Layout read from RM0455 for this device, not from a sibling part.
\begin{tikzpicture}[font=\sffamily\small]
\renewcommand{\regwidth}{8}
\renewcommand{\regunit}{16}

\node[font=\sffamily\bfseries\small, anchor=west] at (0mm,22mm)
  {CRC\_CR, control. Bits 31 down to 8 are reserved and read as zero};
\begin{scope}
\bitscale
\bitfield{7}{7}{regfield}{REV\_OUT}
\bitfield{6}{5}{regfield}{REV\_IN}
\bitfield{4}{3}{regset}{POLYSIZE}
\bitfield{2}{1}{regrsvd}{Res.}
\bitfield{0}{0}{regset}{RESET}
\end{scope}
\node[font=\tiny\ttfamily, anchor=north west] at (0mm,-1mm)
  {this chapter: REV\_OUT = 0, REV\_IN = 00, POLYSIZE = 01, RESET written once
   per message};

\node[font=\sffamily\bfseries\small, anchor=west] at (0mm,-9mm)
  {CRC\_INIT, the value loaded when RESET is written};
\node[regro, minimum width=128mm, text width=124mm, minimum height=7mm,
      anchor=south west] at (0mm,-20mm) {0x0000\,FFFF};

\node[font=\sffamily\bfseries\small, anchor=west] at (0mm,-26mm)
  {CRC\_POL, the polynomial, right aligned in the register};
\node[regro, minimum width=128mm, text width=124mm, minimum height=7mm,
      anchor=south west] at (0mm,-37mm) {0x0000\,1021};

\node[font=\sffamily\scriptsize, anchor=north west, text width=62mm] at (0mm,-42mm)
  {\textbf{POLYSIZE} 00 is 32 bit, 01 is 16 bit, 10 is 8 bit, 11 is 7 bit.\\
   \textbf{REV\_IN} 00 none, 01 by byte, 10 by half word, 11 by word.\\
   \textbf{REV\_OUT} 0 none, 1 bit reversed.\\
   \textbf{RESET} write one to reload CRC\_INIT. It clears itself.};
\node[umlnote, text width=60mm, anchor=north west] at (68mm,-42mm)
  {The two reversal fields are the ones nobody checks, and most published
   16-bit checksums need them non-zero. The reference used here needs neither,
   which is why it is the one to get agreeing first. No setting here fixes the
   other trap: writing the data register as a word feeds four bytes.};
\end{tikzpicture}
